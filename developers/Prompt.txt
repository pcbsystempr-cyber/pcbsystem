create extension if not exists "uuid-ossp";

create table public.programmers (
  id uuid default gen_random_uuid() primary key,
  name text not null,
  role text not null,
  photo text,
  created_at timestamp with time zone default now()
);

create table public.projects (
  id uuid default gen_random_uuid() primary key,
  name text not null,
  programmer text not null,
  description text,
  icon text,
  created_at timestamp with time zone default now()
);

create table public.media (
  id uuid default gen_random_uuid() primary key,
  type text not null,
  src text not null,
  caption text,
  created_at timestamp with time zone default now()
);

alter table public.programmers enable row level security;
alter table public.projects enable row level security;
alter table public.media enable row level security;

create policy "Public read programmers" on public.programmers for select using (true);
create policy "Public insert programmers" on public.programmers for insert with check (true);
create policy "Public update programmers" on public.programmers for update using (true);
create policy "Public delete programmers" on public.programmers for delete using (true);

create policy "Public read projects" on public.projects for select using (true);
create policy "Public insert projects" on public.projects for insert with check (true);
create policy "Public update projects" on public.projects for update using (true);
create policy "Public delete projects" on public.projects for delete using (true);

create policy "Public read media" on public.media for select using (true);
create policy "Public insert media" on public.media for insert with check (true);
create policy "Public update media" on public.media for update using (true);
create policy "Public delete media" on public.media for delete using (true);
